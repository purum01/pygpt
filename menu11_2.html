질문
다음 조건에 맞는 파이썬 코드를 생성해주세요 #제목 파일 월별 정리하기 #조건 1. 현재 작업 폴더 하위의 'scandata' 폴더에 있는 파일들을 월별로 정리 2. 날짜 정보는 파일 이름으로 판별 — 파일 이름 형식은 "년_월_일" 3. scandata 폴더의 파일들을 현재 작업 폴더 하위 'organized' 폴더에 월별 폴더를 만들어 분류 4. 폴더가 없으면 자동 생성 5. 파이썬 입문자가 이해할 수 있도록 주석 포함 #출력예시 organized/ ├── 2024_01/ │ ├── 2024_01_05.jpg │ └── 2024_01_22.jpg └── 2024_02/ └── 2024_02_10.jpg 파일 정리가 완료되었습니다.
답변 

import os
import shutil

# 원본 폴더와 정리 폴더 경로 설정
scandata_folder  = "scandata"
organized_folder = "organized"

# organized 폴더 생성 (이미 있으면 무시)
os.makedirs(organized_folder, exist_ok=True)

# scandata 폴더의 항목들을 순서대로 처리
for file_name in os.listdir(scandata_folder):
    source_path = os.path.join(scandata_folder, file_name)

    # 파일이 아닌 항목(하위 폴더 등)은 건너뜀
    if not os.path.isfile(source_path):
        continue

    # 파일 이름에서 년도와 월 추출 (예: "2024_01_15.jpg" → 2024, 1)
    parts = file_name.split('_')
    year  = parts[0]
    month = parts[1]

    # 월별 폴더 경로 생성 및 폴더 만들기
    month_folder = os.path.join(organized_folder, f"{year}_{int(month):02d}")
    os.makedirs(month_folder, exist_ok=True)

    # 파일을 월별 폴더로 이동
    destination_path = os.path.join(month_folder, file_name)
    shutil.move(source_path, destination_path)

print("파일 정리가 완료되었습니다.")
