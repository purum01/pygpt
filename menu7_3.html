동물원 소리 내기 

def animal_sound(animal):
    if animal == "고양이":
        return "야옹"
    elif animal == "개":
        return "멍멍"
    elif animal == "오리":
        return "꽥꽥"
    else:
        return "저는 그 동물의 소리를 모르겠어요!"

print("고양이가 무슨 소리를 낼까요?")
print(animal_sound("고양이"))   # 야옹

print("\n개가 무슨 소리를 낼까요?")
print(animal_sound("개"))       # 멍멍

print("\n오리가 무슨 소리를 낼까요?")
print(animal_sound("오리"))     # 꽥꽥

print("\n사자가 무슨 소리를 낼까요?")
print(animal_sound("사자"))     # 저는 그 동물의 소리를 모르겠어요!
과일 주스 바 

def make_juice(fruit):
    juice_menu = {
        "사과": "사과 주스가 준비되었습니다!",
        "오렌지": "오렌지 주스가 준비되었습니다!",
        "포도": "포도 주스가 준비되었습니다!",
    }

    if fruit in juice_menu:
        return juice_menu[fruit]
    else:
        return f"죄송해요, {fruit} 주스는 없어요. 다른 과일을 선택해 주세요!"

print(make_juice("사과"))    # 사과 주스가 준비되었습니다!
print(make_juice("오렌지"))  # 오렌지 주스가 준비되었습니다!
print(make_juice("포도"))    # 포도 주스가 준비되었습니다!
print(make_juice("바나나"))  # 죄송해요, 바나나 주스는 없어요. 다른 과일을 선택해 주세요!
온도 변환기 

def print_options():
    print("'c' 섭씨온도 → 화씨온도 변환")
    print("'f' 화씨온도 → 섭씨온도 변환")
    print("'q' 종료")

def celsius_to_fahrenheit(c_temp):
    return 9.0 / 5.0 * c_temp + 32

def fahrenheit_to_celsius(f_temp):
    return (f_temp - 32.0) * 5.0 / 9.0

print_options()
choice = input("메뉴에서 선택하세요: ")
while choice != "q":
    if choice == "c":
        temp = float(input("섭씨온도: "))
        print("화씨온도:", celsius_to_fahrenheit(temp))
    elif choice == "f":
        temp = float(input("화씨온도: "))
        print("섭씨온도:", fahrenheit_to_celsius(temp))
    print_options()
    choice = input("메뉴에서 선택하세요: ")
