import random

def generate_lotto_numbers():
    # 1~45 중 중복 없이 6개 선택 후 오름차순 정렬
    lotto_numbers = random.sample(range(1, 46), 6)
    lotto_numbers.sort()
    return lotto_numbers

def generate_lotto_sets(num_sets=5):
    # num_sets 만큼 로또 번호 세트를 생성하여 리스트로 반환
    lotto_sets = []
    for _ in range(num_sets):
        lotto_sets.append(generate_lotto_numbers())
    return lotto_sets

# 5세트 생성 후 출력
lotto_sets = generate_lotto_sets()

for i, lotto_set in enumerate(lotto_sets, 1):
    print(f"세트 {i}: {lotto_set}")
