'Confirmed' 컬럼명은 파일 이름에서 확장자를 제외한 이름으로 변경, '-'는 '/'로 변환 5. 각 파일을 'Country_Region' 컬럼 기준으로 병합 6. 병합 결과를 Excel 파일(covid_2020.xlsx)로 저장 7. 파이썬 입문자가 이해할 수 있도록 주석 포함 #출력예시 Country_Region 1/22/2020 1/23/2020 ... Afghanistan 0.0 0.0 Albania 0.0 0.0 ... 저장 완료: covid_2020.xlsx

import os
import pandas as pd

PATH = 'covid/'

# CSV 파일만 필터링
file_list = []
for f in os.listdir(PATH):
    if f.endswith('.csv'):
        file_list.append(f)

first_doc = True

for file in file_list:
    # CSV 파일 읽기
    doc = pd.read_csv(PATH + file)

    # 필요한 컬럼만 추출
    doc = doc[['Country_Region', 'Confirmed']]

    # Country_Region 기준으로 그룹핑 후 Confirmed 합계
    doc = doc.groupby('Country_Region').sum()

    # 컬럼명을 파일 이름 기반으로 변경 (예: 01-22-2020.csv → 1/22/2020)
    date_column = file.split('.')[0].lstrip('0').replace('-', '/')
    doc.columns = [date_column]

    # 첫 번째 파일은 기준 DataFrame으로, 이후 파일은 병합
    if first_doc:
        final_doc  = doc
        first_doc  = False
    else:
        final_doc = pd.merge(final_doc, doc, how='outer',
                             left_index=True, right_index=True)

# 병합 결과를 Excel 파일로 저장
final_doc.to_excel('covid_2020.xlsx')
print("저장 완료: covid_2020.xlsx")
